def evaluate(model_wrap, loader, device):
    y_true_all, y_pred_all = [], []
    model_wrap.model.eval()
    with torch.no_grad():
        for x, y in tqdm(loader, desc="[Eval]", leave=False):
            x, y = x.to(device), y.to(device)
            y_hat = model_wrap.sample(x, seq_len=y.shape[1])
            y_true_all.append(y)
            y_pred_all.append(y_hat)
    y_true = torch.cat(y_true_all)
    y_pred = torch.cat(y_pred_all)
    return rmse(y_true, y_pred), pcrr(y_true, y_pred)

import numpy as np
import seaborn as sns
from scipy.ndimage import gaussian_filter, median_filter
import cv2


def interpolate_grid(resolution=100):
    """创建高分辨率网格并准备用于预测的特征"""
    x_grid = np.linspace(0, 1, resolution)
    y_grid = np.linspace(0, 1, resolution)
    xx, yy = np.meshgrid(x_grid, y_grid)
    return pd.DataFrame({"X": xx.flatten(), "Y": yy.flatten()})

df = pd.read_csv(cfg["csv"])
batch_size = cfg.get("batch", 256)


def load_model(cfg: dict, device: str = "cpu"):
    model_wrap = NDP(
        in_dim=cfg["D"],
        time_step=cfg["T"],
        device=device,
        hidden=cfg["hidden"],
        n_layers=cfg["layers"],
    )
    state_dict = torch.load(cfg["ckpt"], map_location=device)
    model_wrap.model.load_state_dict(state_dict)
    model_wrap.model.eval()
    return model_wrap


def apply_smoothing(data):
    data = np.nan_to_num(data, nan=np.nanmean(data))
    data = median_filter(data, size=5)
    data = gaussian_filter(data, sigma=3.0)
    data_norm = (
        (data - np.nanmin(data)) / (np.nanmax(data) - np.nanmin(data)) * 255
    ).astype(np.uint8)
    filtered = cv2.bilateralFilter(data_norm, d=9, sigmaColor=100, sigmaSpace=100)
    filtered = cv2.bilateralFilter(filtered, d=9, sigmaColor=75, sigmaSpace=75)
    data = filtered / 255 * (np.nanmax(data) - np.nanmin(data)) + np.nanmin(data)
    data = gaussian_filter(data, sigma=2.0)
    return data

def evaluate_with_noise(model, test_loader, device, noise_level=0.0):
    """在测试数据上评估模型在指定噪声水平下的性能"""
    y_true_all, y_pred_all = [], []
    model.model.eval()

    with torch.no_grad():
        for x, y in tqdm(
            test_loader, desc=f"[噪声水平={noise_level:.3f}]", leave=False
        ):
            x, y = x.to(device), y.to(device)

            if noise_level > 0:
                # 添加高斯噪声到特征
                noise = torch.randn_like(x) * noise_level
                x_noisy = x + noise
                # 确保噪声后的值仍在合理范围内 [0, 1]
                x_noisy = torch.clamp(x_noisy, 0, 1)
                y_hat = model.sample(x_noisy, 1)
            else:
                y_hat = model.sample(x, 1)

            y_true_all.append(y)
            y_pred_all.append(y_hat)

    y_true = torch.cat(y_true_all)
    y_pred = torch.cat(y_pred_all)

    return {
        "rmse": rmse(y_true, y_pred),
        "pcrr": pcrr(y_true, y_pred),
        "noise_level": noise_level,
    }

results = []

    results.append(result)
    print(f"[噪声={noise_level:.3f}] RMSE={result['rmse']:.5f} | PCRR={result['pcrr']:.5f}")
